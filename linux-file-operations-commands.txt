#!/bin/bash

# ===== Configuration =====
CONTAINER_NAME="ansible_node"
IMAGE_NAME="ubuntu:20.04"
ROOT_PASSWORD="kartik123"  # Change if needed
ANSIBLE_DIR="$HOME/ansible-docker"  # Change if your hosts.ini is elsewhere
INVENTORY_FILE="$ANSIBLE_DIR/hosts.ini"

echo "🔹 Creating Docker container for Ansible..."

# 1️⃣ Ensure Ansible project directory exists
mkdir -p "$ANSIBLE_DIR"

# 2️⃣ Ensure hosts.ini exists
if [ ! -f "$INVENTORY_FILE" ]; then
    echo "[docker_nodes]" > "$INVENTORY_FILE"
fi

# 3️⃣ Pull Ubuntu image if not already present
docker pull $IMAGE_NAME

# 4️⃣ Run container in background
docker run -dit --name $CONTAINER_NAME --hostname ansible-node $IMAGE_NAME

# 5️⃣ Install SSH, sudo, Python inside container
echo "🔹 Installing SSH and Python inside the container..."
docker exec -it $CONTAINER_NAME bash -c "
  apt update &&
  apt install -y openssh-server sudo python3 &&
  echo 'root:$ROOT_PASSWORD' | chpasswd &&
  service ssh start
"

# 6️⃣ Get container IP
CONTAINER_IP=$(docker inspect -f '{{range.NetworkSettings.Networks}}{{.IPAddress}}{{end}}' $CONTAINER_NAME)

# 7️⃣ Remove old entry if exists
sed -i "/$CONTAINER_IP/d" "$INVENTORY_FILE"

# 8️⃣ Add container to Ansible inventory automatically
echo "$CONTAINER_IP ansible_user=root ansible_password=$ROOT_PASSWORD ansible_python_interpreter=/usr/bin/python3" >> "$INVENTORY_FILE"

# 9️⃣ Show final info
echo ""
echo "✅ Container '$CONTAINER_NAME' is ready for Ansible!"
echo "📌 Container IP: $CONTAINER_IP"
echo "📌 Root password: $ROOT_PASSWORD"
echo "📌 Ansible inventory updated: $INVENTORY_FILE"
echo ""
echo "Test connection with:"
echo "ansible -i $INVENTORY_FILE docker_nodes -m ping"
echo ""
echo "Then run your playbooks as usual."
